```yaml
name: Finalize v1.6.0-obiwave.1

on:
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: recover-v1.6.0-obiwave.1
  cancel-in-progress: false

jobs:
  validate:
    permissions:
      contents: read
      packages: read
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-node@v5
        with:
          node-version: '22'
      - uses: docker/setup-buildx-action@v4

        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}
      - name: Verify immutable recovery artifacts
        env:
          GH_TOKEN: ${{ github.token }}
        run: >-
          node scripts/release/recovery-manifest.mjs verify-all
          --manifest security/releases/v1.6.0-obiwave.1.json
          --scanner security/trivy-scanner.json
          --repository ObiWanCanOweMe/obiwave

  vulnerability-policy:
    needs: [validate]
    uses: ./.github/workflows/scan-images.yml
    with:
      release_tag: v1.6.0-obiwave.1
      recovery_manifest: security/releases/v1.6.0-obiwave.1.json
    permissions:
      contents: read
      packages: read
      security-events: write

  deploy-production:
    needs: [validate, vulnerability-policy]
    runs-on: ubuntu-latest
    environment: production
    concurrency:
      group: subwave-production
      cancel-in-progress: false
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-node@v5
        with:
          node-version: '22'
      - name: Deploy immutable release through Portainer
        env:
          PORTAINER_URL: ${{ vars.PORTAINER_URL }}
          PORTAINER_API_KEY: ${{ secrets.PORTAINER_API_KEY }}
          PORTAINER_STACK_ID: ${{ vars.PORTAINER_STACK_ID }}
          PORTAINER_ENDPOINT_ID: ${{ vars.PORTAINER_ENDPOINT_ID }}
          SUBWAVE_RELEASE_TAG: v1.6.0-obiwave.1
          SUBWAVE_HEALTH_URL: ${{ vars.SUBWAVE_HEALTH_URL }}
          SUBWAVE_STREAM_URL: ${{ vars.SUBWAVE_STREAM_URL }}
          SUBWAVE_STREAM_PASSWORD: ${{ secrets.SUBWAVE_STREAM_PASSWORD }}
        run: node scripts/deploy/portainer-release.mjs
```
